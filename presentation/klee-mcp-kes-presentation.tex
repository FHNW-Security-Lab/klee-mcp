% 03 | 01:30--02:30
\begin{frame}{The triage gap}
  \centering\input{figures/triage}
  \vspace{3mm}
  \begin{points}
    \item LLMs propose candidates and infer context
    \item Symbolic execution checks behavior in an explicit model
    \item The harness and its preconditions connect the two
  \end{points}
  \note{[1:00; cumulative 2:30] A candidate is a suspicion, not a confirmed report. A harness is a small program that sets up the function and its inputs for analysis. The hard interface is deciding what the caller can supply and which conditions are valid. Avoid claiming that LLMs can never produce test inputs; the methodological point is that their claims need independent execution evidence.}
\end{frame}

\begin{frame}{A candidate is an explicit specification}
  \centering\input{figures/candidate}
  \vspace{2mm}
  \begin{points}
    \item Make the LLM's assumptions visible and reviewable
    \item Separate the suspected weakness from its input model
  \end{points}
  \note{[1:30; cumulative 6:00] Walk down the four rows. The function and weakness class identify the claim. Input declarations identify what the analysis varies. % 07 | 06:00--07:00
\begin{frame}{Reading the verdict}
  \renewcommand{\arraystretch}{1.6}
  \begin{tabular}{@{}p{31mm}p{101mm}@{}}
    \textbf{Confirmed} & A matching error exists in the analyzed model\\
    \textbf{Infeasible} & No matching error on the fully explored modeled paths\\
    \textbf{Unresolved} & Timeout, build failure, or engine error prevents a conclusion\\
  \end{tabular}
  \vspace{4mm}
  \begin{points}
    \item Check assumptions, completed paths, and engine limitations
    \item A local witness alone does not establish deployment impact
  \end{points}
  \note{[1:00; cumulative 7:00] The server has five literal statuses: confirmed, infeasible, timeout, build\_failed, and klee\_error. Unresolved groups the last three only for this slide. Infeasibility is conditional on sound modeling and completed exploration; a returned status alone is insufficient. Our longjmp examples later show why. Matching error matters: the runner filters unrelated aborts and engine warnings.}
\end{frame}

% 08 | 07:00--08:30
\begin{frame}{Bounds encode a caller contract}
  \centering\input{figures/contracts}
  \vspace{3mm}
  \begin{points}
    \item Bounds select which inputs the analysis considers possible
    \item A safe result inside that set says nothing about excluded inputs
  \end{points}
  \takeaway{Review bounds as part of the claim being verified}
  \note{[1:30; cumulative 8:30] Spend time on this slide: it is the central insight. The outer rectangle is the input space represented by the harness, not every possible real-world execution. The green ellipse is a caller contract, for example that a size argument correctly describes an allocated object. A function can be safe inside the contract and fail outside it. Neither result is contradictory. The diagram is schematic and does not represent measured coverage.}
\end{frame}

% 09 | 08:30--10:00
\begin{frame}{Retracting a questionable assumption}
  \centering\input{figures/relax}
  \vspace{3mm}
  \begin{points}
    \item Retain the initial verdict and record the relaxed retry
    \item A changed verdict exposes sensitivity to the assumption
    \item Review whether the relaxed inputs are possible for real callers
  \end{points}
  \note{[1:30; cumulative 10:00] Describe the deliberately wrong toy bound qualitatively. The first run excludes the error path; the optional retry retracts the bound and finds the error. Explain the scope carefully: automatic relaxation can reveal a bad assumption, but does not prove the relaxed input is realistic. The retry may also time out. Removing declared bounds still leaves the harness, allocation sizes, and engine model in place.}
\end{frame}

\begin{frame}{A historical CVE illustrates the contract}
  \begin{points}
    \item CVE-2015-8540 case study: an internal libpng helper
    \item Function-level analysis reports an error across three versions
    \item Adding the caller contract changes the verdict to infeasible
  \end{points}
  \vspace{4mm}
  \begin{tikzpicture}
    \node[bad,text width=42mm] (a) at (0,0) {\textbf{Contract omitted}\\Local error reported};
    \node[good,text width=42mm] (b) at (7.2,0) {\textbf{Contract supplied}\\No error in explored model};
    \draw[arrow] (a) -- node[above,font=\footnotesize,align=center]{Change the\\precondition} (b);
  \end{tikzpicture}
  \takeaway{Caller-side fixes can leave the helper's precondition essential}
  \note{[1:30; cumulative 15:00] Introduce our historical CVE case study. The extracted helper was studied in libpng 1.2.56, 1.6.19, and 1.6.47. The historical fix was in callers. Observing a local error when bypassing their contract therefore does not establish a newly reachable vulnerability in the later library. Keep the presentation at the evidence and contract level; the scientific conclusion does not need a byte-level trigger.}
\end{frame}

% 14 | 15:00--16:00
\begin{frame}{Reachability changes the interpretation}
  \centering\input{figures/reachability}
  \vspace{3mm}
  \begin{points}
    \item \texttt{png\_format\_number}: an unchecked internal precondition
    \item Shipped callers satisfy the precondition at every reviewed call site
    \item Result: a defense-in-depth finding, with no internal error reachability
  \end{points}
  \note{[1:00; cumulative 16:00] Distinguish this finding from the historical CVE case. We inspected the library call sites: they satisfy the helper's contract. The local error is relevant to defensive hardening but does not establish a remote vulnerability. The reachability tool probes entry-to-function reachability; merely reaching the target function does not prove the error state is reachable under the same calling context. Automated severity labels still need human review.}
\end{frame}

% 15 | 16:00--17:30
\begin{frame}{Where the evidence stops}
  \begin{points}
    \item \textbf{Evaluation scope}: a small, selected benchmark; no external tool comparison
    \item \textbf{Model limits}: parser paths and \texttt{setjmp}/\texttt{longjmp} can limit exploration
    \item \textbf{Caller validity}: relaxing bounds does not establish a realistic execution
    \item \textbf{Impact}: heuristic labels and CVSS prefills need human assessment
  \end{points}
  \vspace{4mm}
  \begin{tikzpicture}
    \draw[arrow] (0,0) -- (11.3,0);
    \foreach \x/\t in {0/Candidate,3.6/Local evidence,7.3/Caller context,11.3/Impact}{
      \fill[teal] (\x,0) circle (2pt);
      \node[above=3mm,font=\small] at (\x,0) {\t};
    }
  \end{tikzpicture}
  \note{[1:30; cumulative 17:30] The evaluation demonstrates a method, not a count of externally acknowledged new vulnerabilities. We had received no maintainer feedback at submission time. The libjpeg case completed only one path because of engine limitations; a superficially reassuring status would be misleading without context. Automatic classification is a triage hint rather than proof of exploitability. The next evaluation should use externally labeled cases, controlled baselines, and realistic caller models.}
\end{frame}

% 16 | 17:30--18:30
\begin{frame}{What this work contributes}
  \begin{points}
    \item \textbf{Explicit candidate schema}\\A checkable claim with declared inputs and preconditions
    \item \textbf{Retractable bounds}\\A second analysis exposes assumption-sensitive verdicts
    \item \textbf{Evidence for human review}\\Execution results, caller context, and reporting artifacts
  \end{points}
  \vspace{3mm}
  \takeaway{The LLM proposes the model; KLEE checks behavior within it}
  \note{[1:00; cumulative 18:30] Position our contribution relative to existing work. Other systems generate harnesses or place an LLM inside a symbolic or concolic engine. Here the LLM remains outside KLEE and commits to a candidate specification. The distinguishing methodological contribution is explicit, retractable preconditions together with reviewable evidence. Do not claim that combining an LLM and symbolic execution is itself new.}
\end{frame}
